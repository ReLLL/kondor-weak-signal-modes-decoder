# Native Linux build. Compile against the selected SDR host's actual headers/core.
enable_language(C)
set(DX_HOST "sdrpp" CACHE STRING "Linux host: sdrpp or brown")
if(NOT DX_HOST MATCHES "^(sdrpp|brown)$")
 message(FATAL_ERROR "Linux requires DX_HOST=sdrpp or brown")
endif()
set(DX_HOST_SOURCE "" CACHE PATH "Source tree matching the Linux SDR host")
set(DX_HOST_CORE "" CACHE FILEPATH "Selected host's libsdrpp_core.so")
set(BROWN_SOURCE "${CMAKE_SOURCE_DIR}/build" CACHE PATH "Pinned MSHV source staged by scripts/stage_sources.py")
if(NOT EXISTS "${DX_HOST_SOURCE}/core/src/imgui/imgui.h" OR NOT EXISTS "${DX_HOST_CORE}")
 message(FATAL_ERROR "Set DX_HOST_SOURCE to matching SDR++ headers and DX_HOST_CORE to its built/installed libsdrpp_core.so. See docs/LINUX.md.")
endif()
option(DX_ADDITIONAL_MODES "Build WSPR/FST4/FST4W/Q65 helpers" ON)
option(DX_JS8 "Build JS8 helper" ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
pkg_check_modules(DSP REQUIRED IMPORTED_TARGET fftw3f volk)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0 pangocairo)
set(DX_COMMON model reception_quality modes sequence_capture date_time table_options country_summary store store_stats store_retention store_cleanup saved_cleanup helper_client engine engine_js8 engine_saved cartography location_check csv_log gridtracker country_distance store_country_distance share_card)
list(TRANSFORM DX_COMMON PREPEND "${CMAKE_SOURCE_DIR}/src/")
list(TRANSFORM DX_COMMON APPEND ".cpp")
add_library(dx_core STATIC ${DX_COMMON})
include("${CMAKE_SOURCE_DIR}/cmake/ShareBackground.cmake")
set_target_properties(dx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dx_core PUBLIC "${CMAKE_SOURCE_DIR}/src" "${DX_HOST_SOURCE}/core/src")
target_compile_definitions(dx_core PUBLIC _GNU_SOURCE DX_HOST_BROWN=$<STREQUAL:${DX_HOST},brown> DX_HOST_SDRSHARP=0 DX_ADDITIONAL_MODES=$<BOOL:${DX_ADDITIONAL_MODES}> DX_JS8=$<BOOL:${DX_JS8}> DX_VERSION="${PROJECT_VERSION}")
target_compile_options(dx_core PRIVATE -Wall -Wextra -Wno-misleading-indentation "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.")
target_link_libraries(dx_core PUBLIC SQLite::SQLite3 Threads::Threads ${CMAKE_DL_LIBS})
set(DECODER "${BROWN_SOURCE}/decoder_modules/ft8_decoder/src")
if(NOT EXISTS "${DECODER}/fftw_mshv_plug.cpp")
 message(FATAL_ERROR "Stage the pinned decoder with scripts/stage_sources.py before configuring. The Linux installer does this automatically.")
endif()
file(GLOB MSHV CONFIGURE_DEPENDS "${DECODER}/ft8_etc/*.cpp" "${DECODER}/ft8_etc/*.c")
add_library(dx_mshv STATIC ${MSHV} "${DECODER}/fftw_mshv_plug.cpp" "${BROWN_SOURCE}/core/src/utils/flog.cpp")
target_include_directories(dx_mshv SYSTEM PUBLIC "${DECODER}" "${BROWN_SOURCE}/core/src")
target_compile_options(dx_mshv PRIVATE -O3 -Wno-register)
target_link_libraries(dx_mshv PUBLIC PkgConfig::DSP Threads::Threads)
add_executable(dx_decoder "${CMAKE_SOURCE_DIR}/src/helper.cpp")
target_link_libraries(dx_decoder PRIVATE dx_mshv dx_core)
set(DX_UI plugin plugin_ui status_bar reset_defaults filter_bar compass uptime station_actions help map settings mode_settings settings_saved tiles corner table_views details country_database share_ui)
list(TRANSFORM DX_UI PREPEND "${CMAKE_SOURCE_DIR}/src/")
list(TRANSFORM DX_UI APPEND ".cpp")
add_library(dx_notebook MODULE ${DX_UI} native.cpp share_native.cpp)
target_include_directories(dx_notebook PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${DX_HOST_SOURCE}/core/src/imgui")
target_link_libraries(dx_notebook PRIVATE dx_core "${DX_HOST_CORE}" PkgConfig::DSP PkgConfig::CURL PkgConfig::GTK OpenGL::GL ${CMAKE_DL_LIBS})
target_compile_definitions(dx_notebook PRIVATE DX_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} DX_VERSION_MINOR=${PROJECT_VERSION_MINOR} DX_VERSION_PATCH=${PROJECT_VERSION_PATCH})
set_target_properties(dx_notebook PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/dx_notebook.host.json" CONTENT "{\"host\":\"${DX_HOST}\",\"version\":\"${PROJECT_VERSION}\",\"platform\":\"Linux\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\"}\n")
include(modes.cmake)
add_executable(dx_host_probe host_probe.cpp)
target_link_libraries(dx_host_probe PRIVATE ${CMAKE_DL_LIBS})
add_executable(dx_share_tests "${CMAKE_SOURCE_DIR}/tests/share_card.cpp" share_native.cpp native.cpp)
target_link_libraries(dx_share_tests PRIVATE dx_core PkgConfig::GTK)
add_test(NAME dx_share_card COMMAND dx_share_tests)
add_test(NAME dx_linux_installer COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_linux_installer.py")
add_executable(dx_table_help_tests "${CMAKE_SOURCE_DIR}/tests/table_help.cpp")
target_include_directories(dx_table_help_tests PRIVATE "${CMAKE_SOURCE_DIR}/src" "${DX_HOST_SOURCE}/core/src/imgui")
add_test(NAME dx_table_help COMMAND dx_table_help_tests)
find_program(XVFB_RUN xvfb-run)
if(XVFB_RUN)
 add_executable(dx_linux_clipboard_tests "${CMAKE_SOURCE_DIR}/tests/linux_clipboard.cpp" native.cpp share_native.cpp)
 target_include_directories(dx_linux_clipboard_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(dx_linux_clipboard_tests PRIVATE dx_core PkgConfig::GTK)
 add_test(NAME dx_linux_clipboard COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:dx_linux_clipboard_tests>)
 set_tests_properties(dx_linux_clipboard PROPERTIES TIMEOUT 30)
endif()
add_executable(dx_fixture "${CMAKE_SOURCE_DIR}/tests/fixture.cpp")
target_link_libraries(dx_fixture PRIVATE dx_mshv dx_core)
add_executable(dx_mock "${CMAKE_SOURCE_DIR}/tests/mock_helper.cpp")
target_link_libraries(dx_mock PRIVATE dx_core)
foreach(test tests table_columns country_summary reception_quality activity_filters history_loading saved_cleanup moderation map_locations reset_defaults retention recovery)
 add_executable(dx_${test}_tests "${CMAKE_SOURCE_DIR}/tests/${test}.cpp")
 target_link_libraries(dx_${test}_tests PRIVATE dx_core)
 add_test(NAME dx_${test} COMMAND dx_${test}_tests $<TARGET_FILE:dx_mock>)
endforeach()
add_test(NAME dx_ft_quality COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_ft_quality.py" $<TARGET_FILE:dx_decoder> $<TARGET_FILE:dx_fixture>)
set_tests_properties(dx_ft_quality PROPERTIES TIMEOUT 180)
install(TARGETS dx_notebook dx_decoder dx_host_probe LIBRARY DESTINATION . RUNTIME DESTINATION .)
foreach(helper dx_modes_decoder dx_wspr_decoder dx_js8_decoder)
 if(TARGET ${helper})
  install(TARGETS ${helper} RUNTIME DESTINATION .)
 endif()
endforeach()
install(FILES "${CMAKE_SOURCE_DIR}/resources/world.geojson" "${CMAKE_SOURCE_DIR}/resources/cty.dat" "${CMAKE_BINARY_DIR}/dx_notebook.host.json" "${CMAKE_SOURCE_DIR}/LICENSE" "${CMAKE_SOURCE_DIR}/CREDITS.md" DESTINATION .)
